func imagesAccepts(args []string) bool {

// runsPs reports whether args reach `ps` under a root that, like nerdctl's,
// sets TraverseChildren and has a persistent --hosts-dir StringSlice.
func runsPs(t *testing.T, args []string) bool {
	ran := false
	root := &cobra.Command{Use: "nerdctl", TraverseChildren: true}
	root.PersistentFlags().StringSlice("hosts-dir", nil, "")
	root.AddCommand(&cobra.Command{Use: "ps", Run: func(*cobra.Command, []string) { ran = true }})
	root.SetOut(io.Discard)
	root.SetArgs(args)
	if err := root.Execute(); err != nil {
		t.Fatal(err)
	}
	return ran
}

Skill improvements ¶

  • When a change rewrites a command line for a program that parses it in more than one stage, such as cobra's subcommand traversal before pflag's flag parse, compare the rewrite under each stage, not only the final one. The stages read the same tokens differently, so a rewrite that the flag parser accepts can still send the earlier stage to another command.
