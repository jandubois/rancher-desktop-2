// allocatedMiB reports the disk space a file occupies, as opposed to its size.
func allocatedMiB(t *testing.T, path string) float64 {
	t.Helper()
	fi, err := os.Stat(path)
	if err != nil {
		t.Fatal(err)
	}
	return float64(fi.Sys().(*syscall.Stat_t).Blocks*512) / (1 << 20)
}

+// sparseBlockSize matches the default 4 KiB allocation block of APFS, ext4,
+// XFS, and btrfs, the smallest hole any of them can make.
+const sparseBlockSize = 4 << 10
+
+var zeroBlock [sparseBlockSize]byte
+
+// sparseWriter writes to f, skipping each block of zeros so that the file
+// system leaves a hole there. An xz stream records no holes, so writing
+// every byte would allocate a disk image's free space along with its data.
+type sparseWriter struct {
+	f       *os.File
+	off     int64 // offset in f of the next byte to write
+	dataEnd int64 // offset in f just past the last data written
+}
+
+func (w *sparseWriter) Write(p []byte) (int, error) {
+	for i := 0; i < len(p); {
+		data := w.runEnd(p, i, false)
+		if data > i {
+			if err := w.writeData(p[i:data], w.off+int64(i)); err != nil {
+				return i, err
+			}
+		}
+		i = w.runEnd(p, data, true)
+	}
+	w.off += int64(len(p))
+	return len(p), nil
+}
+
+// writeData writes data at off in f, then punches a hole over the whole
+// blocks of zeros skipped since the previous data.
+func (w *sparseWriter) writeData(data []byte, off int64) error {
+	if _, err := w.f.WriteAt(data, off); err != nil {
+		return err
+	}
+	holeStart := (w.dataEnd + sparseBlockSize - 1) / sparseBlockSize * sparseBlockSize
+	holeEnd := off / sparseBlockSize * sparseBlockSize
+	if holeEnd > holeStart {
+		punchHole(w.f, holeStart, holeEnd-holeStart)
+	}
+	w.dataEnd = off + int64(len(data))
+	return nil
+}

+// runEnd returns the index in p where the run starting at i ends. The run
+// holds blocks of zeros if zero is true, and blocks of data otherwise. Blocks
+// end at multiples of sparseBlockSize in f, so a block of zeros split across
+// two Write calls still becomes a hole.
+func (w *sparseWriter) runEnd(p []byte, i int, zero bool) int {
+	for i < len(p) {

+		if bytes.Equal(p[i:end], zeroBlock[:end-i]) != zero {

+	}

rdd/docs/design/discovery.md

Ordering every step before the annotation keeps the "ready = clients may proceed" contract consistent. Any client that waits for the annotation sees the CRDs, the enabled-controller list, and working admission webhooks.

The ConfigMap is recreated on every control plane startup, so a stale `ready` from a previous crash is always cleared before CRD installation begins.

The annotation only covers controllers known at control plane startup. External controllers (for example, the mock controller, or a controller manager started later via `rdd svc create --controllers=...`) can attach at any time, so a client waiting for the ready annotation must not assume that every controller it cares about is already registered. Discover external controllers by reading the ConfigMap directly.

Review Process Notes ¶

Skill improvements ¶

  • Expect a liveness monitor to expire. The Monitor tool ends every monitor after 30 minutes, whatever persistent says, so a 90-minute deadline needs its monitor re-armed on each expiry with the deadline's remaining seconds. Step 4's claim that persistent: true lifts the timeout is wrong and should say this instead.
  • In PR mode, check the local branch's reflog as well as its tip before launching agents. A forced gh pr checkout resets the branch to the pushed head, so the tip comparison shows no divergence while an unpushed commit that belongs to the PR survives only in the reflog. List any commit a recent reset dropped, and settle which SHA to review.
  • When a change sends existing input through a different implementation (a library decoder instead of an external tool, a copy loop instead of a platform copy), compare what the two leave on disk, not only the bytes. Sparseness, permissions, ownership, timestamps and extended attributes differ silently between implementations, and a content-equality test passes either way.

Repo context updates ¶

  • [repo] macOS ships GNU make 3.81, whose $(shell) does not see command-line variable assignments while recipes do. Makefile logic that derives the build target with $(shell go env ...) must read $(GOOS) and $(GOARCH) first, or make <target> GOARCH=... builds an artifact whose parts disagree.
  • [repo] The rdd Makefile's recipes also run under a WSL-hosted make that invokes go.exe, and a variable reaches go.exe only when the recipe lists it in WSLENV. Check every variable a new recipe passes to a Go command against that recipe's WSLENV list.
  • [repo] scripts/postinstall.ts runs every dependency, make build-rdd included, under one ten-minute InstallTimeout. A change that adds network work to make build-rdd must fit that limit on a cold cache or give the step its own.
