// server, deletes the three bulk fixture volumes the way the GUI's bulk delete
// does (Promise.all, so concurrently), restores them by creating Volumes of the
// same names as the e2e spec does, and counts the deleted volumes that exist
// again 2s later.
func TestVolumeDeleteProbe(t *testing.T) {
	ctrllog.SetLogger(logr.Discard())
	env := &envtest.Environment{
		CRDInstallOptions:    envtest.CRDInstallOptions{Paths: []string{"crd.yaml"}},
		DownloadBinaryAssets: true,
	}
	cfg, err := env.Start()
	if err != nil {
		t.Fatal(err)
	}
	defer func() { _ = env.Stop() }()

	scheme := runtime.NewScheme()
	_ = clientgoscheme.AddToScheme(scheme)
	_ = containersv1alpha1.AddToScheme(scheme)
	c, err := client.New(cfg, client.Options{Scheme: scheme})
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	for _, ns := range []string{mockNamespaceName, appv1alpha1.AppNamespace} {
		if err := c.Create(ctx, &corev1.Namespace{ObjectMeta: metav1.ObjectMeta{Name: ns}}); err != nil {
			t.Fatal(err)
		}
	}
	mgr, err := ctrl.NewManager(cfg, ctrl.Options{Scheme: scheme, Metrics: metricsserver.Options{BindAddress: "0"}})
	if err != nil {
		t.Fatal(err)
	}
	if err := (&volumeReconciler{Client: mgr.GetClient()}).SetupWithManager(mgr); err != nil {
		t.Fatal(err)
	}
	go func() { _ = mgr.Start(ctx) }()

	list := func() []containersv1alpha1.Volume {
		var volumes containersv1alpha1.VolumeList
		if err := c.List(ctx, &volumes, client.InNamespace(appv1alpha1.AppNamespace)); err != nil {
			t.Fatal(err)
		}
		return volumes.Items
	}
	// waitFor waits until exactly want volumes exist, each with the status
	// that the GUI lists them by.
	waitFor := func(want int) {
		var have []string
		for deadline := time.Now().Add(20 * time.Second); time.Now().Before(deadline); time.Sleep(50 * time.Millisecond) {
			have = nil
			for _, v := range list() {
				if v.Status.Name != "" {
					have = append(have, v.Name)
				}
			}
			if len(have) == want && len(list()) == want {
				return
			}
		}
		t.Fatalf("never reached %d volumes with status; have %v", want, have)
	}
	waitFor(4)

	bulk := []string{"logs", "cache", "backup"}
	del := func(name string) {
		err := c.Delete(ctx, &containersv1alpha1.Volume{ObjectMeta: metav1.ObjectMeta{Name: name, Namespace: appv1alpha1.AppNamespace}})
		if err != nil && !apierrors.IsNotFound(err) {
			t.Error(err)
		}
	}
	restore := func(name string) {
		err := c.Create(ctx, &containersv1alpha1.Volume{ObjectMeta: metav1.ObjectMeta{Name: name, Namespace: appv1alpha1.AppNamespace}})
		if err != nil {
			t.Fatal(err)
		}
	}
	sequential := os.Getenv("PROBE_SEQUENTIAL") != ""
	fastRestore := os.Getenv("PROBE_FAST_RESTORE") != ""
	cameBack := 0
	const rounds = 20
	for round := 0; round < rounds; round++ {
		if sequential {
			for _, name := range bulk {
				del(name)
				time.Sleep(300 * time.Millisecond)
			}
		} else {
			var wg sync.WaitGroup
			for _, name := range bulk {
				wg.Add(1)
				go func() { defer wg.Done(); del(name) }()
			}
			wg.Wait()
		}
		time.Sleep(2 * time.Second)
		existing := map[string]bool{}
		for _, v := range list() {
			existing[v.Name] = true
		}
		for _, name := range bulk {
			if existing[name] {
				cameBack++
				fmt.Printf("round %d: %s exists 2s after its delete\n", round, name)
				del(name) // the second delete marks it again, so the restore starts clean
			}
		}
		waitFor(1)
		if fastRestore {
			for _, name := range bulk {
				restore(name)
			}
			waitFor(4)
			continue
		}
		// Restore one at a time and settled, so only the delete path is measured.
		for i, name := range bulk {
			time.Sleep(300 * time.Millisecond)
			restore(name)
			waitFor(2 + i)
		}
	}
	fmt.Printf("sequential=%v fastRestore=%v: %d of %d deleted volumes came back\n", sequential, fastRestore, cameBack, rounds*len(bulk))
	if cameBack > 0 {
		t.Errorf("%d deleted volumes came back", cameBack)
	}
}

Review Process Notes ¶

Skill improvements ¶

  • Read every probe that an agent ran in its session log, not only the one that its review quotes. An agent can run a realistic probe that refutes its finding, then write a contrived one that confirms it, and report only the second.
  • Before launching agents, run one of the repo's own test commands with the toolchain on PATH. When the repo pins a runtime version that the host default does not meet, stage a compatible one and name it in the prompt, because otherwise every agent's test run fails at load, and the agent takes that failure for a test result.

Repo context updates ¶

  • [repo] The Jest suites need a Node with zlib.createZstdDecompress, which jsdom's bundled undici calls at load: 22.14 lacks it, 22.17 has it, and the repo pins Node 26.13.0 under devEngines. On an older Node every jsdom suite fails at load with Cannot read properties of undefined (reading 'bind'), so treat that as an environment failure, not a test result.
  • [repo] Every e2e spec runs against the mock controller in rdd/pkg/controllers/mock. A mock reconciler that tracks client actions through watch-event handlers races Reconcile, because Reconcile acts on the API server while the handlers see events late. An envtest test in package mock with DownloadBinaryAssets: true, as pkg/controllers/base/finalizer_test.go sets up, reproduces such a race in about a minute without starting rdd.
